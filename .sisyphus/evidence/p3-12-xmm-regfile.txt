  Flags:      0x100 - 0x105 (CF through OF)
  XMM/YMM:    0x400 - 0x5FF (16 slots × 32-byte stride)
    XMMn  = offset 0x400 + n*32, size 16
    YMMn  = offset 0x400 + n*32, size 32
